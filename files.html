function displayFiles(files) { const tbody = document.getElementById('fileTableBody'); let folders = 0; let fileItems = 0; let totalSize = 0; // گروه‌بندی فایل‌های چندتکه const zipParts = {}; const rows = files.map(file => { if (file.type === 'dir') { folders++; // بررسی وجود فایل‌های zip در این پوشه return ` 📁 پوشه ${escapeHtml(file.name)} - 📂 مشاهده در GitHub `; } else { // بررسی فایل‌های zip چندتکه if (file.name.match(/\.(zip|z[0-9]+)$/)) { const baseName = file.name.replace(/\.(zip|z[0-9]+)$/, ''); if (!zipParts[baseName]) zipParts[baseName] = []; zipParts[baseName].push(file); return null; // فعلاً نمایش نده } fileItems++; totalSize += file.size; const icon = getFileIcon(file.name); return ` ${icon} فایل ${escapeHtml(file.name)} ${formatFileSize(file.size)} ⬇️ دانلود مستقیم `; } }).filter(row => row !== null); // اضافه کردن فایل‌های چندتکه به عنوان یک آیتم for (const [baseName, parts] of Object.entries(zipParts)) { const totalPartsSize = parts.reduce((sum, p) => sum + p.size, 0); rows.push(` 📦 فایل چندتکه ${escapeHtml(baseName)} (${parts.length} قطعه) ${formatFileSize(totalPartsSize)} 📂 مشاهده قطعات `); } tbody.innerHTML = rows; updateStats(files.length, folders, fileItems, totalSize); } // تابع جدید برای دانلود و合并 فایل‌های چندتکه async function downloadMultiPartZip(parts) { const zip = new JSZip(); for (const part of parts) { const response = await fetch(part.download_url); const blob = await response.blob(); const fileName = part.name; zip.file(fileName, blob); } const content = await zip.generateAsync({ type: 'blob' }); const link = document.createElement('a'); link.href = URL.createObjectURL(content); link.download = `${parts[0].name.replace(/\.(zip|z[0-9]+)$/, '')}_merged.zip`; link.click(); alert('✅ فایل‌های zip合并 شدند! فایل نهایی را در پوشه دانلودها پیدا کنید.\nسپس با unzip از حالت فشرده خارج کنید.'); } // تابع برای دانلود کل پوشه به صورت ZIP async function downloadFolderAsZip(folderPath, folderName) { const token = document.getElementById('tokenInput')?.value || ''; const apiUrl = `https://api.github.com/repos/${owner}/${repo}/contents/${folderPath}`; try { const response = await fetch(apiUrl, { headers: token ? { 'Authorization': `Bearer ${token}` } : {} }); const items = await response.json(); const zip = new JSZip(); for (const item of items) { if (item.type === 'file') { const fileRes = await fetch(item.download_url, { headers: token ? { 'Authorization': `Bearer ${token}` } : {} }); const blob = await fileRes.blob(); zip.file(item.name, blob); } } const content = await zip.generateAsync({ type: 'blob' }); const link = document.createElement('a'); link.href = URL.createObjectURL(content); link.download = `${folderName}.zip`; link.click(); } catch (error) { alert('خطا: ' + error.message + '\nاگر مخزن خصوصی است، توکن معتبر وارد کنید.'); } }
